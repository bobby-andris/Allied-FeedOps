    ```sql
    SELECT
      COUNT(*) AS total_snapshots,
      COUNT(publish_event_id) AS has_publish_event_id,
      COUNT(content_version) AS has_content_version,
      COUNT(cohort_type) AS has_cohort_type,
      COUNT(days_since_publish) AS has_days_since_publish,
      COUNT(master_sku) AS has_master_sku,
      COUNT(platform) AS has_platform
    FROM performance_snapshots;
    ```

    **Query 3: Join chain completeness (the actual feedback view join)**
    ```sql
    SELECT
      COUNT(*) AS total_snapshots,
      COUNT(ps.publish_event_id) AS snapshots_with_publish_event,
      COUNT(pe.id) AS matched_publish_events,
      COUNT(pe.prompt_hash) AS matched_with_prompt_hash,
      COUNT(pe.content_version) AS matched_with_content_version,
      COUNT(gc.id) AS matched_generated_content
    FROM performance_snapshots ps
    LEFT JOIN publish_events pe ON ps.publish_event_id = pe.id
    LEFT JOIN generated_content gc ON pe.master_sku = gc.master_sku AND pe.platform = gc.platform;
    ```

    **Query 4: performance_impact_scores linkage**
    ```sql
    SELECT
      COUNT(*) AS total_scores,
      COUNT(publish_event_id) AS has_publish_event_id,
      COUNT(master_sku) AS has_master_sku
    FROM performance_impact_scores;
    ```

    **Query 5: Temporal analysis — when did prompt_hash start being populated?**
    ```sql
    SELECT
      DATE_TRUNC('week', published_at) AS week,
      COUNT(*) AS events,
      COUNT(prompt_hash) AS with_prompt_hash,
      ROUND(100.0 * COUNT(prompt_hash) / NULLIF(COUNT(*), 0), 1) AS pct
    FROM publish_events
    WHERE status = 'success'
    GROUP BY 1
    ORDER BY 1;
    ```
