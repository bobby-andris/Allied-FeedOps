export type FallbackLevel = 'per_group' | 'global' | 'defaults'
export type ConfidenceLevel = 'High' | 'Medium' | 'Low'
export interface GroupDistributions {
  customLabel0: string
  tiers: Record<FunnelTier, TierDistribution>
  boundaries: TierBoundaries
  totalTerms: number
  scoredTerms: number
  insufficientTiers: FunnelTier[]
}
export interface TierDistribution {
  tier: FunnelTier; metrics: { roas, cvr, cpc, ctr: MetricDistribution }
  sampleSize: number; fallbackLevel: FallbackLevel
}
export interface MetricDistribution { p25, p50, p75, mean, mad, min, max: number }
export interface TermScore {
  searchTerm, customLabel0, currentTier, recommendedTier: string
  isMisplaced: boolean; tierFitScores: Record<FunnelTier, number>
  confidence: ConfidenceResult; impact: ImpactRange | null
  fallbackLevel: FallbackLevel; verdict, peerContext: string
}
export interface ImpactRange { low, mid, high: number; currency: 'USD'; period: 'monthly' }
export interface ConfidenceResult { score: number; level: ConfidenceLevel; factors: ConfidenceFactors }
