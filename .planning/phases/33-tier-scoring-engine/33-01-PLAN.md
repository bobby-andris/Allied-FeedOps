```typescript
import type { FunnelTier, QueryIntentFeatures } from '@/lib/shopping-funnel/types'

export type FallbackLevel = 'per_group' | 'global' | 'defaults'
export type ConfidenceLevel = 'High' | 'Medium' | 'Low'

export interface TierDistribution {
  tier: FunnelTier
  metrics: {
    roas: MetricDistribution
    cvr: MetricDistribution
    cpc: MetricDistribution
    ctr: MetricDistribution
  }
  sampleSize: number
  fallbackLevel: FallbackLevel
}

export interface MetricDistribution {
  p25: number
  p50: number  // median
  p75: number
  mean: number
  mad: number  // median absolute deviation
  min: number
  max: number
}

export interface GroupDistributions {
  customLabel0: string
  tiers: Record<FunnelTier, TierDistribution>
  boundaries: TierBoundaries
  totalTerms: number
  scoredTerms: number
  insufficientTiers: FunnelTier[]  // tiers with <5 non-zero-metric terms
}

export interface TierBoundaries {
  highFloor: BoundaryValue   // MEDIUM p25 — above this = HIGH candidate
  lowCeiling: BoundaryValue  // MEDIUM p75 — below this = LOW candidate
  metric: 'roas'             // primary metric for boundaries
}

export interface BoundaryValue {
  value: number
  capped: boolean           // was the shift capped at 15%?
  uncappedValue: number     // what the data-driven value was before capping
  previousValue: number | null
}

export interface TermScore {
  searchTerm: string
  customLabel0: string
  currentTier: FunnelTier
  recommendedTier: FunnelTier
  isMisplaced: boolean
  tierFitScores: Record<FunnelTier, number>  // robust z-score per tier
  confidence: ConfidenceResult
  impact: ImpactRange | null  // null if not misplaced
  fallbackLevel: FallbackLevel
  verdict: string             // plain English explanation
  peerContext: string         // e.g., "ranks in top 15% of Towel Bar terms"
}

export interface ConfidenceResult {
  score: number        // 0-1 combined score
  level: ConfidenceLevel
  factors: ConfidenceFactors
}

export interface ConfidenceFactors {
  dataVolume: number       // 0-1, weight 30%
  consistency: number      // 0-1, weight 30%
  significance: number     // 0-1, weight 20%
  intentAlignment: number  // 0-1, weight 20%
}

export interface ImpactRange {
  low: number    // conservative (p25 scenario)
  mid: number    // expected (p50 scenario)
  high: number   // optimistic (p75 scenario)
  currency: 'USD'
  period: 'monthly'
}

export interface ScoringResult {
  distributions: Map<string, GroupDistributions>
  scores: TermScore[]
  globalFallback: Record<FunnelTier, TierDistribution>
  computedAt: string
  totalGroups: number
  totalTermsScored: number
  totalMisplaced: number
  totalImpact: ImpactRange
  heroCallout: string  // "23 terms may be in the wrong tier — $2.4K/mo potential impact"
}
```

**GREEN phase: implement all computation functions in `tier-scoring.ts`.**

Import from `simple-statistics`: `median`, `medianAbsoluteDeviation`, `quantile`, `mean`, `standardDeviation`, `interquartileRange`.

**Functions to implement:**

1. **`computeTierDistributions(rows: LabelTierPerformance[], options?: { previousDistributions?: Map }): Map<string, GroupDistributions>`**
   - Group rows by `custom_label_0`
   - For each group, partition by `tier` (HIGH/MEDIUM/LOW)
   - For each tier partition: extract ROAS/CVR/CPC/CTR arrays
     - ROAS = `row.roas`
     - CVR = `row.conversions / Math.max(row.clicks, 1)`
     - CPC = `(row.cost_micros / 1_000_000) / Math.max(row.clicks, 1)`
     - CTR = `row.clicks / Math.max(row.impressions, 1)`
   - Cap ROAS at p99 before computing distributions (use `quantile(values, 0.99)`)
   - Compute p25/p50/p75/mean/mad/min/max for each metric
   - If tier has <5 rows with non-zero metrics (clicks > 0 OR conversions > 0), mark as insufficient and set `fallbackLevel: 'per_group'` still but add tier to `insufficientTiers`
   - Compute boundaries via `computeTierBoundaries`
   - Also compute global fallback distributions (all groups combined)

2. **`computeGlobalDistributions(rows: LabelTierPerformance[]): Record<FunnelTier, TierDistribution>`**
   - Same as per-group but aggregated across all groups
   - fallbackLevel always 'global'

3. **`computeTierBoundaries(mediumTier: TierDistribution, previousBoundaries?: TierBoundaries): TierBoundaries`**
   - `highFloor` = MEDIUM tier ROAS p25 (terms above this are HIGH candidates)
   - `lowCeiling` = MEDIUM tier ROAS p75 (terms below this are LOW candidates)
   - Cap shift at 15% of previous value (`MAX_BOUNDARY_SHIFT_PERCENT = 0.15`)
   - Track capped/uncappedValue/previousValue

4. **`scoreTerm(term: ExistingFunnelTerm, groupDist: GroupDistributions, globalFallback: Record<FunnelTier, TierDistribution>, intentFeatures?: QueryIntentFeatures): TermScore`**
   - Determine current tier from `term.funnels[0].tier` (map 'High' → 'HIGH' etc.)
   - Determine current `custom_label_0` from `term.funnels[0].custom_label_0`
   - Compute term metrics: ROAS = `total_conversions_value / Math.max(total_cost_micros / 1_000_000, 0.01)`, CVR, CPC, CTR
   - Choose distribution: if group tier is in `insufficientTiers`, use `globalFallback`; else use `groupDist.tiers[tier]`
   - Robust z-score per tier: `(termROAS - tierMedianROAS) / tierMAD` (handle MAD=0 by returning 0)
   - `tierFitScores`: compute fit for each tier (higher = better fit for that tier)
   - `recommendedTier`: tier with highest fit score
   - `isMisplaced`: recommendedTier !== currentTier
   - Generate `verdict` string: "This term is a [strong/moderate/weak] fit for [TIER] because [reason]"
   - Generate `peerContext` string: compute percentile rank within group

5. **`computeConfidence(term, intentFeatures?, tierSemantics?): ConfidenceResult`**
   - Data volume (30%): `Math.min(term.total_clicks / 100, 1)`
   - Consistency (30%): Since we lack daily ROAS, use coefficient of variation proxy: `1 - Math.min(stddev(term metrics across funnels) / mean, 1)`. If only one funnel assignment, use 0.5 (neutral).
   - Statistical significance (20%): `Math.min(term.total_conversions / 10, 1)`
   - NLP alignment (20%): If `intentFeatures` provided, score based on: branded terms align with HIGH (brand queries convert well), competitor terms align with LOW (defensive), product-specific terms align with MEDIUM. Without intent features, use 0.5 (neutral).
   - Combined = weighted sum; level = >=0.70 High, >=0.40 Medium, <0.40 Low

6. **`estimateImpact(term, currentDist, targetDist): ImpactRange`**
   - Monthly impressions from `term.total_impressions` (assume 30-day window)
   - CVR delta range: low = target p25 CVR - current p50 CVR, mid = target p50 - current p50, high = target p75 - current p50
   - AOV: use `$85` default for Allied Brass
   - low/mid/high = `max(0, impressions * cvrDelta * aov)`
   - Return with `currency: 'USD', period: 'monthly'`

7. **`getCachedDistributions(rows, previousDistributions?): { distributions, globalFallback }`**
   - Module-level cache with 10-minute TTL
   - Return cached if fresh, recompute if stale

8. **`buildHeroCallout(scores): string`**
   - Count misplaced terms
   - Sum conservative impact (low bounds)
   - Format: "X terms may be in the wrong tier — $Y/mo potential impact"
