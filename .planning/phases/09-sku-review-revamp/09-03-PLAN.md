    ```typescript
    import { createClient } from '@/lib/supabase/server'
    import { NextResponse } from 'next/server'

    export async function POST(request: Request) {
      const { master_sku, platform } = await request.json()
      if (!master_sku || !platform) {
        return NextResponse.json({ error: 'master_sku and platform required' }, { status: 400 })
      }

      const supabase = createClient()

      // For google/bing: mark the generated_content rows as approved by copying candidate_content to approved_content
      // For shopify: same pattern
      // This is a "quick approve" — it approves whatever candidate content exists for this platform
      const { error: contentError } = await supabase
        .from('generated_content')
        .update({ approved_content: supabase.rpc('coalesce_candidate', {}) })
        .eq('master_sku', master_sku)
        .eq('platform', platform)
        .is('approved_content', null)
        .not('candidate_content', 'is', null)

      // Use a direct update approach — copy candidate_content to approved_content for rows where approved_content is null
      // Supabase doesn't support column-to-column copy in .update(), so use raw SQL via rpc or two queries
      // Approach: fetch candidate then update
      const { data: contentRows } = await supabase
        .from('generated_content')
        .select('id, candidate_content')
        .eq('master_sku', master_sku)
        .eq('platform', platform)
        .is('approved_content', null)
        .not('candidate_content', 'is', null)

      if (contentRows && contentRows.length > 0) {
        for (const row of contentRows) {
          await supabase
            .from('generated_content')
            .update({ approved_content: row.candidate_content })
            .eq('id', row.id)
        }
      }

      // Update sku_approvals to reflect approval
      const approvalUpdate: Record<string, boolean | string> = {}
      if (platform === 'google' || platform === 'bing') {
        approvalUpdate.title_approved = true
        approvalUpdate.description_approved = true
      } else if (platform === 'shopify') {
        approvalUpdate.title_approved = true
        approvalUpdate.description_approved = true
        approvalUpdate.image_approved = true
      }
      approvalUpdate.approval_status = 'approved'

      await supabase
        .from('sku_approvals')
        .upsert({ master_sku, ...approvalUpdate }, { onConflict: 'master_sku' })

      return NextResponse.json({ success: true, master_sku, platform })
    }
    ```

    NOTE: The `contentError` variable from the first `.update()` call is not used — remove that dead code line. The correct pattern is the fetch-then-loop approach shown above.

    **`handleQuickApprove` function** — defined in ReviewListClient:
    ```typescript
    const handleQuickApprove = async (masterSku: string, platform: string) => {
      // Optimistic update
      setOptimisticApprovals(prev => ({
        ...prev,
        [masterSku]: new Set([...(prev[masterSku] ?? []), platform]),
      }))

      try {
        const res = await fetch('/api/review/approve-platform', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ master_sku: masterSku, platform }),
        })
        if (!res.ok) {
          // Rollback optimistic update on failure
          setOptimisticApprovals(prev => {
            const updated = new Set(prev[masterSku] ?? [])
            updated.delete(platform)
            return { ...prev, [masterSku]: updated }
          })
          console.error('Approval failed:', await res.text())
        }
      } catch (e) {
        // Rollback on network error
        setOptimisticApprovals(prev => {
          const updated = new Set(prev[masterSku] ?? [])
          updated.delete(platform)
          return { ...prev, [masterSku]: updated }
        })
      }
    }
    ```
