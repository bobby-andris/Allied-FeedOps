    Create `dashboard/src/app/api/monitoring/sync-jobs/route.ts`:
    ```typescript
    import { createClient } from '@/lib/supabase/server'
    import { NextResponse } from 'next/server'

    export async function GET() {
      try {
        const supabase = createClient()

        // Get active job first (status = 'running' or 'pending')
        const activeResult = await supabase
          .from('search_query_sync_jobs')
          .select('id, status, job_type, days_lookback, queries_fetched, queries_enriched, error_message, created_at, completed_at')
          .in('status', ['running', 'pending'])
          .order('created_at', { ascending: false })
          .limit(1)

        // Get last 5 completed/failed jobs
        const historyResult = await supabase
          .from('search_query_sync_jobs')
          .select('id, status, job_type, days_lookback, queries_fetched, queries_enriched, error_message, created_at, completed_at')
          .in('status', ['completed', 'failed'])
          .order('created_at', { ascending: false })
          .limit(5)

        return NextResponse.json({
          active: activeResult.data || [],
          history: historyResult.data || [],
        })
      } catch (error) {
        console.error('Sync jobs fetch error:', error)
        return NextResponse.json({ error: 'Failed to fetch sync jobs' }, { status: 500 })
      }
    }
    ```
