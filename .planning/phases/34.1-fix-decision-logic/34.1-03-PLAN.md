Add a new action case `'label_block'` in the POST switch/dispatch:

```typescript
case 'label_block': {
  const { customLabel0, metadata } = body
  if (!customLabel0) return NextResponse.json({ error: 'customLabel0 required' }, { status: 400 })

  // Upsert label-level block using sentinel search_term
  const { error } = await supabase
    .from('routing_recommendations')
    .upsert({
      search_term: '__LABEL_BLOCK__',
      custom_label_0: customLabel0,
      recommended_action: 'label_block',
      action_scope: 'label',
      review_status: 'accepted',
      accepted: true,
      accepted_at: new Date().toISOString(),
      accepted_by: 'dashboard_user',
      confidence: 1.0,
      metadata: {
        ...metadata,
        history: [{ action: 'label_block', at: new Date().toISOString(), by: 'dashboard_user' }]
      }
    }, { onConflict: 'search_term,custom_label_0' })

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })
  return NextResponse.json({ success: true, action: 'label_block', customLabel0 })
}
```

Also add a section in the POST handler for `'identify_search_candidates'`:
```typescript
case 'identify_search_candidates': {
  // Find high-ROAS, high-volume terms from query_value_scores
  // Criteria: ROAS > 3.0, impressions > 100, conversions > 0
  const { data: candidates, error } = await supabase
    .from('query_value_scores')
    .select('search_term, custom_label_0, model_inputs')
    .not('model_inputs->roas', 'is', null)

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })

  // Filter and insert candidates
  const searchCandidates = (candidates || [])
    .filter(c => {
      const inputs = c.model_inputs as Record<string, number> | null
      return inputs && (inputs.roas ?? 0) > 3.0 && (inputs.impressions ?? 0) > 100 && (inputs.conversions ?? 0) > 0
    })
    .map(c => ({
      search_term: c.search_term,
      custom_label_0: c.custom_label_0,
      recommended_search_tier: 'exact' as const,
      status: 'candidate',
      confidence: 0.8,
      metadata: { source: 'tier_scoring_engine', identified_at: new Date().toISOString() }
    }))

  if (searchCandidates.length > 0) {
    const { error: insertError } = await supabase
      .from('search_buildout_recommendations')
      .upsert(searchCandidates, { onConflict: 'search_term' })
    if (insertError) return NextResponse.json({ error: insertError.message }, { status: 500 })
  }

  return NextResponse.json({ success: true, candidateCount: searchCandidates.length })
}
```
