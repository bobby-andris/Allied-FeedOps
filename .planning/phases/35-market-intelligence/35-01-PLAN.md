```typescript
import { median } from 'simple-statistics'
import type { BcgQuadrant, LongTailBucket, MonthlySearchVolume } from './types'
import { LONG_TAIL_BUCKETS, SEASONAL_THRESHOLD } from './constants'

export function classifyQuadrant(
  roas: number,
  revenue: number,
  medianRoas: number,
  medianRevenue: number
): BcgQuadrant {
  if (roas >= medianRoas && revenue >= medianRevenue) return 'star'
  if (roas >= medianRoas && revenue < medianRevenue) return 'cashCow'
  if (roas < medianRoas && revenue >= medianRevenue) return 'questionMark'
  return 'dog'
}

export function computeMedians(values: number[]): number {
  if (values.length === 0) return 0
  return median(values)
}

export function computeTrendDirection(change: number): 'up' | 'down' | 'flat' {
  if (change > 5) return 'up'
  if (change < -5) return 'down'
  return 'flat'
}

export function classifySeasonalDirection(changePercent: number): 'spiking' | 'declining' | 'stable' {
  if (changePercent > SEASONAL_THRESHOLD) return 'spiking'
  if (changePercent < -SEASONAL_THRESHOLD) return 'declining'
  return 'stable'
}

export function getWordCount(queryText: string): number {
  return queryText.trim().split(/\s+/).length
}

export function getLongTailBucketLabel(wordCount: number): string {
  const bucket = LONG_TAIL_BUCKETS.find(b => wordCount >= b.min && wordCount <= b.max)
  return bucket?.label ?? '5+ words'
}

export function parseMonthlySearchVolumes(raw: unknown): MonthlySearchVolume[] {
  if (!raw) return []
  // monthly_searches JSONB may be stored as text string per project conventions
  const parsed = typeof raw === 'string' ? JSON.parse(raw) : raw
  if (!Array.isArray(parsed)) return []
  return parsed.map((entry: { year?: number; month?: number; monthly_searches?: number }) => ({
    year: entry.year ?? 0,
    month: entry.month ?? 0,
    searches: entry.monthly_searches ?? 0,
  }))
}

export function computeMoMChange(volumes: MonthlySearchVolume[]): { current: number; prior: number; changePercent: number } {
  if (volumes.length < 2) return { current: 0, prior: 0, changePercent: 0 }
  // Sort descending by year/month
  const sorted = [...volumes].sort((a, b) => b.year - a.year || b.month - a.month)
  const current = sorted[0].searches
  const prior = sorted[1].searches
  const changePercent = prior > 0 ? ((current - prior) / prior) * 100 : 0
  return { current, prior, changePercent }
}

export function matchesCompetitor(queryText: string, tokens: readonly string[]): string[] {
  const lower = queryText.toLowerCase()
  return tokens.filter(token => lower.includes(token))
}

export function isBrandTerm(queryText: string, brandTokens: readonly string[]): boolean {
  const lower = queryText.toLowerCase()
  return brandTokens.some(token => lower.includes(token))
}

export function costMicrosToDollars(micros: number): number {
  return micros / 1_000_000
}

export function buildLongTailBuckets(
  terms: Array<{ queryText: string; roas: number; cvr: number; impressions: number; conversions: number; revenue: number; spend: number }>
): LongTailBucket[] {
  const bucketMap = new Map<string, { terms: typeof terms }>()

  for (const bucket of LONG_TAIL_BUCKETS) {
    bucketMap.set(bucket.label, { terms: [] })
  }

  for (const term of terms) {
    const wc = getWordCount(term.queryText)
    const label = getLongTailBucketLabel(wc)
    bucketMap.get(label)?.terms.push(term)
  }

  return LONG_TAIL_BUCKETS.map(bucket => {
    const items = bucketMap.get(bucket.label)?.terms ?? []
    const totalImpressions = items.reduce((s, t) => s + t.impressions, 0)
    const totalConversions = items.reduce((s, t) => s + t.conversions, 0)
    const totalRevenue = items.reduce((s, t) => s + t.revenue, 0)
    const totalSpend = items.reduce((s, t) => s + t.spend, 0)
    return {
      wordCountRange: bucket.label,
      termCount: items.length,
      avgRoas: totalSpend > 0 ? totalRevenue / totalSpend : 0,
      avgCvr: items.length > 0 ? totalConversions / Math.max(items.reduce((s, t) => s + (t.impressions > 0 ? t.impressions : 0), 0), 1) : 0,
      totalImpressions,
      totalConversions,
      totalRevenue,
      totalSpend,
    }
  })
}
```

GET handler that accepts optional `?customLabel0=...` filter. Runs 5 parallel Supabase queries:

1. **Impression Share Gaps (DEMAND-01)**:
   - Query `search_queries` joined with `keyword_metrics` on `LOWER(query_text) = LOWER(keyword)`
   - Aggregate impressions per query_text, compare to avg_monthly_searches
   - Compute sharePercent = (actual / market) * 100
   - Sort by market volume DESC NULLS LAST, limit 100

2. **CPC Opportunity (DEMAND-02)**:
   - From same join, compute actual CPC = cost_micros / clicks
   - Compare to high_cpc_micros from keyword_metrics
   - Headroom = (1 - actual/market) * 100
   - Classify: below_market (headroom > 10%), at_market (-10% to 10%), above_market (headroom < -10%)
   - Sort by headroom DESC, limit 100

3. **Seasonal Patterns (DEMAND-03)**:
   - Query keyword_metrics for terms that have monthly_searches JSONB
   - Parse monthly_searches using parseMonthlySearchVolumes()
   - Compute MoM change using computeMoMChange()
   - Flag terms with |change| > 20% using classifySeasonalDirection()
   - Return only flagged terms (spiking or declining), sorted by |changePercent| DESC

4. **New Term Discovery (DEMAND-04)**:
   - Query search_queries where period_start >= (now - 7 days)
   - Exclude terms that exist with period_start < (now - 7 days)
   - Use a subquery: `AND NOT EXISTS (SELECT 1 FROM search_queries sq2 WHERE sq2.query_text = sq.query_text AND sq2.period_start < CURRENT_DATE - 7)`
   - Since Supabase client doesn't support NOT EXISTS easily, use raw SQL via `.rpc()` or a two-step approach:
     a. Fetch all terms from last 7 days
     b. Fetch all terms from before 7 days ago (just query_text)
     c. Filter client-side: new = in recent but NOT in older
   - Sort by impressions DESC, limit 50

5. **Long-Tail Analysis (DEMAND-07)**:
   - Query search_queries aggregated by query_text
   - For each term, compute word count, ROAS, CVR
   - Use buildLongTailBuckets() to bucket into 1-2, 3-4, 5+
   - Return 3 bucket summaries

Compute KPI summary cards:
- avgImpressionShare: average sharePercent across all terms with market data
- avgCpcHeadroom: average headroomPercent across all terms with market data
- seasonalAlertCount: count of spiking + declining terms
- newTermCount: count of new terms

Return `DemandData` JSON.

Create `dashboard/src/app/api/market-intelligence/products/route.ts`:

GET handler with optional `?customLabel0=...` for filtering and `?group=...` for detail drill-down.

**Default mode (no group param) — BCG overview**:

1. Query `query_value_scores` grouped by `custom_label_0`:
   - Per group: count terms, sum impact_score
   - Extract currentTier from model_inputs JSONB

2. Query `search_queries` grouped by custom_label_0 (join via query_text):
   - Per group: sum impressions, clicks, cost_micros, conversions, conversion_value
   - Compute ROAS = sum(conversion_value) / sum(cost_micros / 1_000_000)
   - Revenue = sum(conversion_value)
   - Spend = sum(cost_micros) / 1_000_000

   **NOTE**: search_queries doesn't have custom_label_0 directly. Join through query_value_scores which has both search_term and custom_label_0. Query pattern:
   ```sql
   SELECT qvs.custom_label_0,
          COUNT(DISTINCT qvs.search_term) as term_count,
          SUM(sq.impressions) as impressions,
          SUM(sq.clicks) as clicks,
          SUM(sq.cost_micros) as cost_micros,
          SUM(sq.conversions) as conversions,
          SUM(sq.conversion_value) as revenue
   FROM query_value_scores qvs
   JOIN search_queries sq ON LOWER(sq.query_text) = LOWER(qvs.search_term)
   GROUP BY qvs.custom_label_0
   ```
   Since Supabase client can't do this join easily, use two queries and merge client-side:
   a. Query query_value_scores grouped by custom_label_0 (term counts)
   b. For each group, query search_queries matching those terms
   OR use a Supabase RPC function / raw SQL if available

   **Simpler approach**: Query search_queries joined with query_value_scores via a single SQL query using the admin client or Supabase SQL. If that's not feasible, do the join in memory:
   - Fetch all query_value_scores (search_term, custom_label_0)
   - Fetch all search_queries (query_text, metrics)
   - Build a Map<customLabel0, aggregatedMetrics> by matching search_term to query_text

3. Compute trend per group using `funnel_snapshots_daily`:
   - Last 30 days revenue vs prior 30 days revenue per custom_label_0
   - Trend = ((recent - prior) / prior) * 100
   - Use computeTrendDirection()

4. Compute medians:
   - medianRoas = computeMedians(groups.map(g => g.roas))
   - medianRevenue = computeMedians(groups.map(g => g.revenue))

5. Classify each group using classifyQuadrant()

6. Compute KPIs:
   - starCount, cashCowCount, questionMarkCount, dogCount
   - totalRevenue, totalSpend

Return `ProductsData` JSON.

2. For each term, fetch metrics from search_queries
3. Sort by impressions DESC, limit 50
