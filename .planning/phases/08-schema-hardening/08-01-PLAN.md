```sql
-- CTE dedup pattern (keep newest row)
WITH ranked AS (
  SELECT id, ROW_NUMBER() OVER (
    PARTITION BY master_sku, platform, environment, snapshot_date
    ORDER BY fetched_at DESC NULLS LAST, id DESC
  ) AS rn
  FROM performance_snapshots
)
DELETE FROM performance_snapshots p USING ranked r WHERE p.id = r.id AND r.rn > 1;

-- Idempotent constraint guard pattern
DO $$ BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'constraint_name' AND conrelid = 'table'::regclass
  ) THEN ALTER TABLE ... ADD CONSTRAINT ...; END IF;
END $$;
```
